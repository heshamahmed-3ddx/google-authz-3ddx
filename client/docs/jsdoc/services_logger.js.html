/**
 * Get current timestamp in YYYY-MM-DD HH:mm:ss format
 * @returns {string} Formatted timestamp
 */
function getTimestamp() {
  const now = new Date();
  const year = now.getFullYear();
  const month = String(now.getMonth() + 1).padStart(2, '0');
  const day = String(now.getDate()).padStart(2, '0');
  const hours = String(now.getHours()).padStart(2, '0');
  const minutes = String(now.getMinutes()).padStart(2, '0');
  const seconds = String(now.getSeconds()).padStart(2, '0');
  
  return `${year}-${month}-${day} ${hours}:${minutes}:${seconds}`;
}

/**
 * Get caller file and line information
 * @returns {Object} Object with filename and line number
 */
function getCallerInfo() {
  const error = new Error();
  const stack = error.stack.split('\n');
  const callerLine = stack[3] || stack[2]; // Get caller's line
  
  // Extract filename and line number
  const match = callerLine.match(/([^/\\]+\.(js|vue|ts)):(\d+):\d+/);
  if (match) {
    const filename = match[1].replace('.js', '').replace('.vue', '').replace('.ts', '');
    const lineNumber = match[3];
    return { filename, lineNumber };
  }
  
  return { filename: 'unknown', lineNumber: '0' };
}

/**
 * Check if a log message is critical enough to log
 * @param {string} message - Log message
 * @param {string} level - Log level
 * @returns {boolean} True if should log
 */
function isCriticalLog(message, level) {
  if (level === 'error' || level === 'warn') return true;
  
  const criticalKeywords = [
    'auth', 'login', 'logout', 'error', 'failed', 'success',
    'api', 'request', 'response', 'navigation', 'route',
    'user', 'session', 'token', 'permission', 'access'
  ];
  
  const messageLower = message.toLowerCase();
  return criticalKeywords.some(keyword => messageLower.includes(keyword));
}

/**
 * Create a client logger with [CLIENT] prefix and colors
 * @param {string} context - Context or component name
 * @returns {Object} Logger object with info, warn, error methods
 */
export function createClientLogger(context = 'App') {
  // Color constants
  const colors = {
    CLIENT: '\x1b[34m',    // Blue for [CLIENT]
    ERROR: '\x1b[31m',     // Red
    WARN: '\x1b[33m',      // Yellow
    INFO: '\x1b[36m',      // Cyan
    DEBUG: '\x1b[35m',     // Magenta
    RESET: '\x1b[0m'       // Reset
  };

  /**
   * Format log message with [CLIENT] prefix and colors
   * @param {string} level - Log level
   * @param {string} message - Log message
   * @param {Object} meta - Additional metadata
   * @returns {string} Formatted log message
   */
  function formatMessage(level, message, meta = {}) {
    const timestamp = getTimestamp();
    const { filename, lineNumber } = getCallerInfo();
    
    // For production builds or non-dev environments, use plain format
    if (import.meta.env.PROD) {
      return `[CLIENT]|${timestamp}|${level.toUpperCase()}|[${filename}:${lineNumber}]|${message}`;
    }
    
    // For development, use colored format
    const clientColor = colors.CLIENT;
    const levelColor = colors[level.toUpperCase()] || colors.INFO;
    const reset = colors.RESET;
    
    return `${clientColor}[CLIENT]${reset}|${timestamp}|${levelColor}${level.toUpperCase()}${reset}|[${filename}:${lineNumber}]|${message}`;
  }

  return {
    /**
     * Log info messages (only critical ones)
     * @param {string} message - Log message
     * @param {Object} meta - Additional metadata
     */
    info: (message, meta = {}) => {
      // Client-side logging disabled
    },

    /**
     * Log warning messages
     * @param {string} message - Log message
     * @param {Object} meta - Additional metadata
     */
    warn: (message, meta = {}) => {
      // Client-side logging disabled
    },

    /**
     * Log error messages
     * @param {string} message - Log message
     * @param {Object} meta - Additional metadata
     */
    error: (message, meta = {}) => {
      // Client-side logging disabled
    },

    /**
     * Log debug messages (only in development)
     * @param {string} message - Log message
     * @param {Object} meta - Additional metadata
     */
    debug: (message, meta = {}) => {
      // Client-side logging disabled
    }
  };
}
