/**
 * Theme Store for managing application themes
 */
export const useThemeStore = defineStore('theme', () => {
  // State
  const currentTheme = ref('light')
  const isAutoTheme = ref(false) // Auto-detect system preference
  
  // Getters
  const isDark = computed(() => currentTheme.value === 'dark')
  const isLight = computed(() => currentTheme.value === 'light')
  const themeDisplayName = computed(() => themeConfig[currentTheme.value]?.displayName || 'Unknown Theme')
  const themeColors = computed(() => themeConfig[currentTheme.value]?.colors || {})
  
  // Available themes
  const availableThemes = computed(() => [
    { value: 'light', label: 'Light', icon: 'mdi-white-balance-sunny' },
    { value: 'dark', label: 'Dark', icon: 'mdi-moon-waning-crescent' }
  ])
  
  // Actions
  
  /**
   * Set the current theme
   * @param {string} theme - Theme name ('light' or 'dark')
   */
  const setTheme = (theme) => {
    if (themeConfig[theme]) {
      currentTheme.value = theme
      isAutoTheme.value = false
      
      // Save to localStorage
      localStorage.setItem('3ddiagnostix-theme', theme)
      localStorage.setItem('3ddiagnostix-auto-theme', 'false')
      
      console.log(`🎨 Theme changed to: ${theme}`)
    } else {
      console.warn(`⚠️ Unknown theme: ${theme}`)
    }
  }
  
  /**
   * Toggle between light and dark themes
   */
  const toggleTheme = () => {
    const newTheme = currentTheme.value === 'light' ? 'dark' : 'light'
    setTheme(newTheme)
  }
  
  /**
   * Enable automatic theme detection based on system preference
   */
  const enableAutoTheme = () => {
    isAutoTheme.value = true
    localStorage.setItem('3ddiagnostix-auto-theme', 'true')
    
    // Apply system preference
    detectSystemTheme()
    
    console.log('🔄 Auto theme detection enabled')
  }
  
  /**
   * Disable automatic theme detection
   */
  const disableAutoTheme = () => {
    isAutoTheme.value = false
    localStorage.setItem('3ddiagnostix-auto-theme', 'false')
    
    console.log('🔒 Auto theme detection disabled')
  }
  
  /**
   * Toggle automatic theme detection
   */
  const toggleAutoTheme = () => {
    if (isAutoTheme.value) {
      disableAutoTheme()
    } else {
      enableAutoTheme()
    }
  }

  /**
   * Detect and apply system theme preference
   */
  const detectSystemTheme = () => {
    if (typeof window !== 'undefined' && window.matchMedia) {
      const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches
      const systemTheme = prefersDark ? 'dark' : 'light'
      
      if (isAutoTheme.value) {
        currentTheme.value = systemTheme
        console.log(`🔍 System theme detected: ${systemTheme}`)
      }
      
      return systemTheme
    }
    return 'light'
  }
  
  /**
   * Initialize theme from localStorage or system preference
   */
  const initializeTheme = () => {
    try {
      // Check localStorage for saved preferences
      const savedTheme = localStorage.getItem('3ddiagnostix-theme')
      const savedAutoTheme = localStorage.getItem('3ddiagnostix-auto-theme')
      
      // Set auto theme preference
      isAutoTheme.value = savedAutoTheme === 'true'
      
      if (isAutoTheme.value) {
        // Use system preference
        detectSystemTheme()
      } else if (savedTheme && themeConfig[savedTheme]) {
        // Use saved theme
        currentTheme.value = savedTheme
      } else {
        // Default to light theme
        currentTheme.value = 'light'
      }
      
      console.log(`🎨 Theme initialized: ${currentTheme.value} (auto: ${isAutoTheme.value})`)
    } catch (error) {
      console.warn('⚠️ Failed to initialize theme from localStorage:', error)
      currentTheme.value = 'light'
    }
  }
  
  /**
   * Get theme color by key
   * @param {string} colorKey - Color key from theme config
   * @returns {string} Color value
   */
  const getThemeColor = (colorKey) => {
    return themeColors.value[colorKey] || colorKey
  }
  
  /**
   * Get CSS custom properties for current theme
   * @returns {Object} CSS custom properties object
   */
  const getCSSProperties = () => {
    const colors = themeColors.value
    const cssProps = {}
    
    Object.entries(colors).forEach(([key, value]) => {
      cssProps[`--color-${key}`] = value
    })
    
    return cssProps
  }
  
  // Listen for system theme changes
  if (typeof window !== 'undefined' && window.matchMedia) {
    const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)')
    mediaQuery.addListener((e) => {
      if (isAutoTheme.value) {
        const systemTheme = e.matches ? 'dark' : 'light'
        currentTheme.value = systemTheme
        console.log(`🔄 System theme changed: ${systemTheme}`)
      }
    })
  }
  
  return {
    // State
    currentTheme,
    isAutoTheme,
    
    // Getters
    isDark,
    isLight,
    themeDisplayName,
    themeColors,
    availableThemes,
    
    // Actions
    setTheme,
    toggleTheme,
    toggleAutoTheme,
    enableAutoTheme,
    disableAutoTheme,
    detectSystemTheme,
    initializeTheme,
    getThemeColor,
    getCSSProperties
  }
})
